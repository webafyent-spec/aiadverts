Home improvement is one of the easiest categories in the world to advertise, and one of the most badly advertised. The product is a transformation: a dated kitchen becomes a showpiece, a leaking roof becomes a sealed one, a bare roof becomes a solar array that keeps the lights on. Nothing sells better than a before and after. Yet most solar installers, builders and renovators in South Africa market themselves with a logo, a phone number and a few blurry site photos.

It's not that the proof doesn't exist. It's sitting in the camera roll of whoever ran the job, scattered across dozens of projects, never turned into anything a potential customer would stop scrolling for.

## Why Trades Businesses Struggle With Video

The obstacles are practical, not a lack of good work to show:

- **The work happens on site**, in clients' homes, surrounded by dust, tools and people who are there to finish a job, not film it
- **Nobody has time** to plan a shoot around an installation schedule, so the "after" shot is usually a quick phone photo on the way out
- **Before shots are rarely taken properly**, which leaves the most persuasive part of the story missing
- **A professional shoot costs more than many jobs are worth** to market, so it never happens
- **Homeowners are cautious.** Letting a contractor into your home is a trust decision, and trust comes from seeing finished work that looks professional

(function () {
  var root = document.querySelector('[data-sb]');
  if (!root) return;
  var screen = root.querySelector('.sb-screen');
  var shotEl = root.querySelector('.sb-shot');

  var reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  var DUR = [3, 4, 5, 3];
  var TOTAL = 15;
  var START = [0, 3, 7, 12];
  var LABELS = ['Hook', 'Problem', 'Reveal', 'Next step'];
  var MOODS = ['hook', 'before', 'after', 'cta'];

    ]
  };

  var trade = 'solar';
  var t = 0;
  var seg = -1;
  var playing = false;
  var last = 0;
  var raf = 0;
  var autoplayed = false;

  function fmt(s) { return '0:' + (s < 10 ? '0' : '') + Math.floor(s); }
  function segAt(time) {
    for (var i = START.length - 1; i >= 0; i--) if (time >= START[i]) return i;
    return 0;
  }
  function showShot(i) {
    var s = trades[trade][i];
    screen.setAttribute('data-mood', MOODS[i]);
    shotEl.textContent = 'Shot ' + (i + 1) + ' · ' + LABELS[i];
    lineEl.textContent = s[0];
    noteEl.textContent = s[1];
    for (var k = 0; k < segs.length; k++) {
      segs[k].classList.toggle('is-current', k === i);
      segs[k].setAttribute('aria-current', k === i ? 'step' : 'false');
    }
    screen.classList.remove('sb-in');
    void screen.offsetWidth;
    screen.classList.add('sb-in');
  }
  function draw(force) {
    head.style.left = (t / TOTAL * 100) + '%';
    timeEl.textContent = fmt(t) + ' / 0:15';
    var i = segAt(t);
    if (i !== seg || force) { seg = i; showShot(i); }
  }
  function setPlaying(on) {
    playing = on;
    root.classList.toggle('is-playing', on);
    icon.textContent = on ? '❚❚' : (t >= TOTAL ? '↻' : '▶');

    cancelAnimationFrame(raf);
    if (on) { last = performance.now(); raf = requestAnimationFrame(tick); }
  }
  function tick(now) {
    t += (now - last) / 1000;
    last = now;
    if (t >= TOTAL) { t = TOTAL; draw(); setPlaying(false); return; }
    draw();
    raf = requestAnimationFrame(tick);
  }

  playBtn.addEventListener('click', function () {
    if (!playing && t >= TOTAL) { t = 0; draw(true); }
    setPlaying(!playing);
  });
  for (var k = 0; k < segs.length; k++) {
    segs[k].addEventListener('click', function () {

      }
      t = 0;
      draw(true);
      if (!playing) icon.textContent = '▶';
    });
  }

  if (typeof IntersectionObserver !== 'undefined') {
    new IntersectionObserver(function (entries) {
      var visible = entries[0].isIntersecting;
      if (visible && !autoplayed && !reduce) { autoplayed = true; setPlaying(true); }
      else if (!visible && playing) setPlaying(false);
    }, { threshold: 0.6 }).observe(screen);
  }

  draw(true);
